SELECT DISTINCT child.id_kid
FROM people person
JOIN relationships parent ON person.id = parent.id_kid
JOIN relationships child USING(id_parent)
JOIN people sibling ON child.id_kid = sibling.id
WHERE person."Фамилия_И._О." = 'Седых В. А.'
  AND child.id_kid != person.id
  AND sibling."пол" = 'М'

Answer: 2599
